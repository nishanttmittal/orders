    // ---- Orders (allowlist-locked: only Google users added in Admin → Users) ----
    // TIGHTENED 2026-10-08 (owner: "anshul ji cannot delete anything only modify it and that modification will be
    // available for me to check"; Day Report / Logs / Money are owner-only). What the rules now guarantee, whatever
    // the app or anyone with a staff login tries:
    //   • only the OWNER can delete anything, cancel an order, or change price / advance
    //   • a manager can create orders (as himself, without money), correct them and mark dispatch, but can never
    //     remove a line, change the order number or whose order it is
    //   • customer / item names: staff may ADD, only the owner may change or delete
    //   • the change log is write-once, stamped with the signer's own email, and readable by the owner only
    //   • the order-number counter can only go up
    // The laptop jobs use the Admin SDK and are not affected by these rules.
    function oPath()  { return /databases/$(database)/documents/apps/orders/users/$(tEmail()); }
    function oUser()  { return signedIn() && (bootstrapOwner() || (tEmail() != '' && exists(oPath()) && get(oPath()).data.active == true)); }
    function oOwner() { return signedIn() && (bootstrapOwner() || (tEmail() != '' && exists(oPath()) && get(oPath()).data.active == true && get(oPath()).data.role == 'owner')); }
    // manager or owner (the people who may post to the staff group and correct orders)
    function oManager() { return signedIn() && (bootstrapOwner() || (tEmail() != '' && exists(oPath()) && get(oPath()).data.active == true && get(oPath()).data.role in ['owner', 'manager'])); }
    function oUnchanged(keys) { return !request.resource.data.diff(resource.data).affectedKeys().hasAny(keys); }

    // users allowlist: only people ON the list (and the owner) may read it — a stranger with a Google account gets
    // nothing, and the app shows him "No access"; only the owner writes
    match /apps/orders/users/{id} { allow read: if oUser(); allow write: if oOwner(); }

    // the staff-group line of an order ("mirror"): staff may only ask for a FAILED line to be sent again — never mark
    // one as sent, never touch what the laptop job recorded
    function oResendOk(n, o) {
      return o.get('mirror', {}).get('status', 'none') in ['held', 'failed', 'uncertain']
             && n.mirror.status == 'pending'
             && n.mirror.diff(o.get('mirror', {})).affectedKeys().hasOnly(['status', 'retry', 'why']);
    }
    // correction / added-line notes for the staff group: ONE new note at the end, signed by the person, waiting to go;
    // notes already there (sent or not) cannot be touched, so a sent note can never be made to go again
    function oNoteOk(n, o) {
      let was = o.get('groupNotes', []);
      let last = n.groupNotes[n.groupNotes.size() - 1];
      return n.groupNotes.size() == was.size() + 1
             && (was.size() == 0 || n.groupNotes[0:was.size()] == was)
             && last.keys().hasOnly(['id', 'kind', 'lines', 'status', 'at', 'by'])
             && last.by == tEmail() && last.status == 'pending' && last.kind in ['ADD', 'EDIT']
             && last.lines is list && last.lines.size() >= 1 && last.lines.size() <= 40;
    }
    function oNewOrderOk(d) {
      let m = d.get('mirror', {});
      let ms = m.get('status', 'none');
      return d.createdByEmail == tEmail()
             && d.get('price', 0) == 0 && d.get('advance', 0) == 0
             && d.status == 'pending' && d.get('source', 'app') == 'app'
             && d.orderNo is string && d.orderNo.matches('^UO-[0-9]{4,6}$')
             && int(d.orderNo.split('-')[1]) < get(/databases/$(database)/documents/apps/orders/meta/counter).data.next
             && d.clientName is string && d.clientName.size() <= 80
             && d.items is list && d.items.size() >= 1 && d.items.size() <= 40
             && !d.keys().hasAny(['groupNotes', 'notePending', 'cancelledAt', 'cancelledBy', 'cancelReason', 'test', 'scanKey', 'scanRef'])
             && m.keys().hasOnly(['status'])
             && (ms == 'none' ? oUser() : (ms == 'pending' && oManager()));
    }
    function oOrderEditOk(n, o, ch) {
      return ch.hasOnly(['clientName', 'items', 'status', 'groupNotes', 'notePending', 'editedAt', 'editedBy', 'lastDispatchAt', 'mirror'])
             && o.get('status', 'pending') != 'cancelled'
             && n.get('status', 'pending') in ['pending', 'production', 'ready', 'dispatched']
             && n.items is list && n.items.size() >= o.get('items', []).size() && n.items.size() <= 40
             && n.clientName is string && n.clientName.size() <= 80
             && (!ch.hasAny(['mirror']) || oResendOk(n, o))
             && (!ch.hasAny(['groupNotes']) || oNoteOk(n, o))
             && (!ch.hasAny(['notePending']) || n.notePending == true)
             && oManager();
    }

    match /apps/orders/orders/{id} {
      allow read: if oUser();
      // a new order is always stamped with the signer's own email; only the owner may enter money; only a manager /
      // owner may ask for a line in the staff group; the number must be one the shared counter has already given out
      allow create: if oOwner() || oNewOrderOk(request.resource.data);
      // manager: may correct an order and enter dispatch — only the listed fields, never the number, whose order it
      // is, money, dates or remarks; never cancel; never end up with fewer lines than before.
      // LIMIT (rules cannot loop over a list): the rule checks the NUMBER of lines, not what is in each line. A manager
      // may rewrite a line's item or quantity — that is the "modify" the owner allows. What makes every such change
      // visible is not this rule but the twice-daily audit snapshot taken by the laptop job (Admin SDK), which compares
      // every order with the copy before and writes its own log entries that no app login can write or alter.
      allow update: if oOwner() || oOrderEditOk(request.resource.data, resource.data, request.resource.data.diff(resource.data).affectedKeys());
      allow delete: if oOwner();
    }

    // customer and item names: staff add, only the owner changes or deletes
    match /apps/orders/clients/{id}  { allow read: if oUser(); allow create: if oManager() && request.resource.data.name is string && request.resource.data.name.size() <= 80; allow update, delete: if oOwner(); }
    match /apps/orders/products/{id} { allow read: if oUser(); allow create: if oManager() && request.resource.data.name is string && request.resource.data.name.size() <= 120; allow update, delete: if oOwner(); }

    // change log: write-once, stamped with the signer's own email; only the owner reads it
    match /apps/orders/logs/{id} {
      allow read: if oOwner();
      // `at` must be the SERVER's clock (so an entry cannot be back-dated), and only the known fields are accepted
      allow create: if oUser() && request.resource.data.get('byEmail', '') == tEmail()
                    && request.resource.data.get('at', null) == request.time
                    && request.resource.data.keys().hasOnly(['id', 'ts', 'at', 'action', 'detail', 'by', 'ref', 'byEmail'])
                    && request.resource.data.get('action', '') is string && request.resource.data.get('action', '').size() <= 40
                    && request.resource.data.get('detail', '') is string && request.resource.data.get('detail', '').size() <= 2000
                    && (oOwner() || request.resource.data.get('by', '') != 'owner')
                    // these two kinds are written only by the laptop jobs (Admin SDK): no phone login may write one
                    && !(request.resource.data.get('action', '') in ['AUDIT', 'PAD_CORRECTION']);
      allow update, delete: if false;
    }

    // doubts: written by the laptop job (Admin SDK); manager / owner settle them, nobody deletes
    match /apps/orders/doubts/{id} {
      allow read: if oManager();
      allow create: if oOwner();
      allow update: if oOwner() || (oManager() && oUnchanged(['customer', 'question', 'source', 'orderId', 'orderNo', 'createdAt', 'lines']));
      allow delete: if oOwner();
    }

    // order-number counter: the number can only go up
    match /apps/orders/meta/{id} {
      allow read: if oUser();
      // the number moves forward a little at a time: it can never go back, and never be thrown far ahead
      allow create: if oUser() && id == 'counter' && request.resource.data.next is int && request.resource.data.next >= 2 && request.resource.data.next <= 50;
      allow update: if oUser() && id == 'counter' && request.resource.data.next is int && request.resource.data.next > resource.data.next && request.resource.data.next <= resource.data.next + 20;
      allow delete: if false;
    }

    // anything else under apps/orders (old WhatsApp inbox, future collections): owner only
    match /apps/orders/{collection}/{docId} {
      allow read, write: if oOwner() && !(collection in ['users', 'orders', 'clients', 'products', 'logs', 'doubts', 'meta']);
    }
    match /apps/orders { allow read: if oUser(); allow write: if oOwner(); }
